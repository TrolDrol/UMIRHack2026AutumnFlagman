type UUID = string;
type Timestamp = string;
type Quantity = string;
type Unit = "g" | "ml" | "pcs";
type Meta = {serverTime: Timestamp; requestId: UUID};
type Envelope<T> = {data: T; meta: Meta};
type Page<T> = {items: T[]; page: number; pageSize: number; total: number};
type User = {id: UUID; username: string; displayName: string; isSuperAdmin?: boolean};
type Role = "super_admin" | "owner" | "manager" | "warehouse" | "waiter";
type Enterprise = {id: UUID; name: string; warehouseId: UUID; createdAt: Timestamp};
type Membership = {
  enterpriseId: UUID; enterpriseName: string; warehouseId: UUID; role: Role;
};
type Member = {userId: UUID; username: string; displayName: string; role: Role};
type Config = {
  restaurantName: string; timeZone: "Europe/Moscow";
  expiryWarningPercent: 10; hotOfferMaxPortions: 20; contractVersion: "2.0.0";
};
type Ingredient = {
  id: UUID; name: string; unit: Unit; defaultShelfLifeHours: number;
  reorderPoint: Quantity; targetStock: Quantity;
};
type IngredientInput = Omit<Ingredient,"id">;
type IngredientPatch = Partial<Omit<IngredientInput,"unit">>;
type RecipeLine = {ingredientId: UUID; quantity: Quantity};
type Dish = {
  id: UUID; name: string; active: boolean; recipe: RecipeLine[];
  createdAt: Timestamp; updatedAt: Timestamp;
};
type DishInput = Pick<Dish,"name"|"active"|"recipe">;
type DeliveryLineInput =
  | {ingredientId: UUID; quantity: Quantity; expiresAt: Timestamp}
  | {ingredientId: UUID; quantity: Quantity; manufacturedAt: Timestamp;
     shelfLifeHours?: number};
type DeliveryInput = {
  number: string; expectedAt: Timestamp|null; note: string|null;
  lines: DeliveryLineInput[];
};
type DeliveryItem = {
  id: UUID; ingredientId: UUID; ingredientName: string; unit: Unit;
  quantity: Quantity; manufacturedAt: Timestamp|null;
  shelfLifeHours: number; expiresAt: Timestamp;
};
type Batch = {
  id: UUID; deliveryId: UUID; deliveryNumber: string; deliveryItemId: UUID;
  ingredientId: UUID; ingredientName: string; unit: Unit;
  receivedAt: Timestamp; manufacturedAt: Timestamp|null;
  shelfLifeHours: number; expiresAt: Timestamp;
  initialQuantity: Quantity; remainingQuantity: Quantity;
  status: "fresh"|"expiring"|"expired"|"empty"; remainingSeconds: number;
};
type Delivery = {
  id: UUID; number: string; createdAt: Timestamp;
  expectedAt: Timestamp|null; receivedAt: Timestamp|null;
  status: "pending"|"confirmed"|"rejected";
  source: "manual"|"integration"; externalId: string|null;
  note: string|null; lines: DeliveryItem[]; batches: Batch[];
  confirmedByUserId: UUID|null; rejectedAt: Timestamp|null; rejectionReason: string|null;
};
type InventoryRow = {
  ingredientId: UUID; ingredientName: string; unit: Unit;
  physicalQuantity: Quantity; availableQuantity: Quantity;
  expiringQuantity: Quantity; expiredQuantity: Quantity;
  nearestExpiresAt: Timestamp|null;
  reorderPoint: Quantity; targetStock: Quantity; isLowStock: boolean;
};
type Requirement = {
  ingredientId: UUID; ingredientName: string; unit: Unit;
  requiredQuantity: Quantity; availableQuantity: Quantity; missingQuantity: Quantity;
};
type Allocation = {batchId: UUID; ingredientId: UUID; quantity: Quantity; expiresAt: Timestamp};
type OrderInput = {lines: {dishId: UUID; portions: number}[]};
type OrderPreview = {canCreate: boolean; requirements: Requirement[]; allocations: Allocation[]};
type Order = {
  id: UUID; number: string; createdAt: Timestamp; status: "completed";
  source: "staff"|"waiter"|"integration"; externalId: string|null;
  lines: {dishId: UUID; dishName: string; portions: number}[]; allocations: Allocation[];
};
type DishAvailability = {
  dishId: UUID; dishName: string; active: boolean; maxPortions: number;
  status: "available"|"stopped"|"inactive"; shortages: Requirement[];
};
type HotOfferReason = {
  ingredientId: UUID; ingredientName: string; unit: Unit; batchIds: UUID[];
  usableExpiringQuantity: Quantity; plannedUsage: Quantity; earliestExpiresAt: Timestamp;
};
type HotOffer = {
  dishId: UUID; dishName: string; score: number; maxPortions: number;
  suggestedPortions: number; earliestExpiresAt: Timestamp;
  reasons: HotOfferReason[]; explanation: string;
};
type PurchaseSuggestion = {
  ingredientId: UUID; ingredientName: string; unit: Unit;
  availableQuantity: Quantity; expiringQuantity: Quantity; planningQuantity: Quantity;
  reorderPoint: Quantity; targetStock: Quantity; targetQuantity: Quantity;
  suggestedQuantity: Quantity;
  reasons: ("below_minimum"|"expiring_stock"|"stopped_dish")[]; affectedDishIds: UUID[];
};
type WriteOffInput = {
  batchId: UUID; quantity: Quantity; reason: "expired"|"damaged"; note: string|null;
};
type WriteOff = WriteOffInput & {id: UUID; createdAt: Timestamp};
type Movement = {
  id: UUID; createdAt: Timestamp; ingredientId: UUID; ingredientName: string;
  unit: Unit; batchId: UUID; type: "receipt"|"order"|"write_off";
  quantityDelta: string; remainingAfter: Quantity;
  referenceId: UUID; referenceNumber: string|null;
};
type Dashboard = {
  ingredientCount: number; activeDishCount: number; stoppedDishCount: number;
  lowStockIngredientCount: number; expiringBatchCount: number; expiredBatchCount: number;
  todayOrderCount: number; hotOfferCount: number; pendingDeliveryCount: number;
};
// Без рецептур, партий, закупочных порогов и статистики склада.
type ServiceMenuItem = {
  dishId: UUID; dishName: string; maxPortions: number;
  status: "available"|"stopped"; unavailableReason: string|null;
};
type ServiceOffer = {
  dishId: UUID; dishName: string; suggestedPortions: number; maxPortions: number;
};
type ServiceShortage = {ingredientName: string; unit: Unit; missingQuantity: Quantity};
type ServicePreview = {canCreate: boolean; shortages: ServiceShortage[]};
type OrderReceipt = Pick<Order,"id"|"number"|"createdAt"|"status"|"lines">;
type IntegrationClient = {
  id: UUID; warehouseId: UUID; name: string; kind: "supplier"|"pos"; createdAt: Timestamp;
};
type IntegrationKey = {
  id: UUID; clientId: UUID; label: string; createdAt: Timestamp;
  expiresAt: Timestamp|null; revokedAt: Timestamp|null;
};
type ApiError = {
  error: {
    code: string; message: string;
    details: {
      fields?: {path: string; message: string}[];
      shortages?: Requirement[]|ServiceShortage[];
    };
  };
  meta: Meta;
};
